👥 4️⃣ register.php

Registers new admins or users + logs registration activity.

<?php
include('dbconn.php');

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $name = $_POST['name'];
    $email = $_POST['email'];
    $password = password_hash($_POST['password'], PASSWORD_DEFAULT);
    $type = $_POST['type']; // admin or user
    $ip = $_SERVER['REMOTE_ADDR'];

    $table = ($type == 'admin') ? 'admin' : 'user';
    $sql = "INSERT INTO $table (name, email, password) VALUES ('$name', '$email', '$password')";

    if (mysqli_query($conn, $sql)) {
        $id = mysqli_insert_id($conn);
        mysqli_query($conn, "INSERT INTO activity_log (user_type, user_id, action, ip_address)
                             VALUES ('$type', '$id', 'register', '$ip')");
        echo "✅ Registration successful! <a href='index.php'>Login</a>";
    } else {
        echo "❌ Error: " . mysqli_error($conn);
    }
}
?>

<?php

$r_admin = mysqli_query($conn, $sql_admin);

if (mysqli_num_rows($r_admin) > 0) {

        $_SESSION['role'] = 'admin';

                             VALUES ('admin', '{$admin['id']}', 'login', '$ip')");
        header("Location: dashboard.php");
        exit;
    }
}

// Check User Table
$sql_user = "SELECT * FROM user WHERE email='$email'";
$r_user = mysqli_query($conn, $sql_user);

if (mysqli_num_rows($r_user) > 0) {
    $user = mysqli_fetch_assoc($r_user);
    if (password_verify($password, $user['password'])) {
        $_SESSION['id'] = $user['id'];
        $_SESSION['role'] = 'user';
        mysqli_query($conn, "INSERT INTO activity_log (user_type, user_id, action, ip_address)
                             VALUES ('user', '{$user['id']}', 'login', '$ip')");
